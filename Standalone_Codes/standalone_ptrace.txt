#!/usr/bin/env python3
"""
Standalone ptrace sensor using tracepoint.
Detects process injection attempts (PTRACE_ATTACH/SEIZE).
"""

from bcc import BPF
import os
import sys
import resource

# --- C Code ---
BPF_CODE = """
#include <uapi/linux/ptrace.h>
#include <uapi/linux/limits.h>
#include <linux/sched.h>

struct ptrace_event_t {
    u32 pid;
    u32 uid;
    char comm[TASK_COMM_LEN];
    u32 tracee_pid;
    u64 request;
};

BPF_PERF_OUTPUT(events);

static inline bool should_ignore() {
    if (bpf_get_current_pid_tgid() >> 32 == 0) return true;
    return false;
}

TRACEPOINT_PROBE(syscalls, sys_enter_ptrace) {
    if (should_ignore()) return 0;

    u64 pid_tgid = bpf_get_current_pid_tgid();
    u32 pid = pid_tgid >> 32;
    u32 uid = bpf_get_current_uid_gid() & 0xFFFFFFFF;

    u64 request = args->request;
    
    // Filter: Only alert on ATTACH (16) or SEIZE (0x4206)
    if (request != 16 && request != 0x4206) return 0;

    struct ptrace_event_t event = {};
    event.pid = pid;
    event.uid = uid;
    event.request = request;
    event.tracee_pid = args->pid;
    bpf_get_current_comm(&event.comm, sizeof(event.comm));

    events.perf_submit(args, &event, sizeof(event));
    return 0;
}
"""

def print_event(cpu, data, size):
    event = b["events"].event(data)
    comm = event.comm.decode('utf-8', errors='ignore')
    req_type = "ATTACH" if event.request == 16 else "SEIZE"
    
    print(f"\033[91m[ALERT]\033[0m PTRACE DETECTED! PID:{event.pid} ({comm}) attaching to {event.tracee_pid} [{req_type}]")
    print(f"  >> Potential Process Injection Attempt")

if __name__ == "__main__":
    if os.geteuid() != 0:
        print("Error: Must run as root")
        sys.exit(1)
        
    resource.setrlimit(resource.RLIMIT_MEMLOCK, (resource.RLIM_INFINITY, resource.RLIM_INFINITY))

    try:
        b = BPF(text=BPF_CODE)
        print("Monitoring ptrace syscalls... (Try: strace -p 1)")
        b["events"].open_perf_buffer(print_event)
        
        while True:
            b.perf_buffer_poll(timeout=100)
    except KeyboardInterrupt:
        print("\nStopped.")
    except Exception as e:
        print(f"Error: {e}")
        sys.exit(1)
